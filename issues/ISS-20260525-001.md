### 修复代码设计

```python
class PaperAnalysisAgent:
    def __init__(
        self,
        registry: ToolRegistry,
        memory: Optional[PaperMemoryManager] = None,
        llm_client: Optional[DashScopeClientSync] = None,  # 新增
        enable_llm: bool = False  # 默认关闭，向后兼容
    ):
        self.llm_client = llm_client
        self.enable_llm = enable_llm
    
    def _generate_llm_summary(self, query: str, context: str) -> str:
        """LLM生成通俗解释"""
        prompt = f"""
        论文内容: {context[:3000]}
        用户问题: {query}
        请用通俗语言解释，用餐巾纸级别的简洁表述。
        """
        
        result = self.llm_client.chat(
            messages=[{"role": "user", "content": prompt}],
            model="glm-5",
            max_tokens=500
        )
        
        # 审计记录
        self.audit.log_llm_call(
            provider="dashscope",
            model="glm-5",
            input_tokens=len(prompt)/4,
            output_tokens=len(result)/4
        )
        
        return result.output
```
