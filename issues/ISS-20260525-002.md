```python
class QualityAssuranceAgent:
    
    def compare_with_source(self, llm_output: str, retrieved_chunks: List[Dict]) -> Dict:
        """对比LLM输出与论文原文"""
        
        issues = {
            "additions": [],      # LLM添加的论文中没有的信息
            "omissions": [],      # 论文有但LLM遗漏的信息
            "distortions": []     # LLM扭曲的信息
        }
        
        # 1. 检查LLM添加的内容
        for entity in self._extract_entities(llm_output):
            if not self._entity_in_chunks(entity, retrieved_chunks):
                issues["additions"].append({
                    "entity": entity,
                    "risk": "potential_hallucination"
                })
        
        # 2. 检查遗漏的关键信息
        key_points = self._extract_key_points(retrieved_chunks)
        for point in key_points:
            if point not in llm_output:
                issues["omissions"].append({
                    "point": point,
                    "suggestion": "建议补充"
                })
        
        # 3. 审计记录
        self.audit.log_hallucination(
            issues=issues,
            support_rate=len(issues["additions"]) / total_entities
        )
        
        return issues
```
